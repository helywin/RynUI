cmake_minimum_required(VERSION 3.25)
project(RynUILibdecorResolverContract LANGUAGES C)

if(NOT DEFINED TEST_SOURCE_DIR)
    message(FATAL_ERROR "TEST_SOURCE_DIR is required.")
endif()

if(TEST_CASE STREQUAL "system")
    set(RYNUI_DEPENDENCY_MODE SYSTEM CACHE STRING "" FORCE)
elseif(TEST_CASE STREQUAL "non-linux")
    set(RYNUI_DEPENDENCY_MODE BUNDLED CACHE STRING "" FORCE)
    set(CMAKE_SYSTEM_NAME Windows)
else()
    set(RYNUI_DEPENDENCY_MODE BUNDLED CACHE STRING "" FORCE)
endif()
include("${TEST_SOURCE_DIR}/cmake/RynUIDependencies.cmake")
rynui_resolve_bundled_libdecor()

if(TEST_CASE STREQUAL "system" OR TEST_CASE STREQUAL "non-linux")
    if(TARGET RynUI::LibDecor OR TARGET rynui_libdecor_external)
        message(FATAL_ERROR "${TEST_CASE} must not create bundled libdecor targets.")
    endif()
else()
    if(NOT TARGET RynUI::LibDecor OR NOT TARGET rynui_libdecor_external)
        message(FATAL_ERROR "Bundled libdecor canonical and external targets are required.")
    endif()
    if(NOT RYNUI_LIBDECOR_STAGE_PREFIX MATCHES "_deps/rynui-libdecor-stage$")
        message(FATAL_ERROR "Bundled libdecor staging prefix is not build-local.")
    endif()
    if(NOT RYNUI_LIBDECOR_BUILD_RPATH STREQUAL
            "${RYNUI_LIBDECOR_STAGE_PREFIX}/lib")
        message(FATAL_ERROR "Bundled libdecor build RPATH must use the staging prefix.")
    endif()
endif()
