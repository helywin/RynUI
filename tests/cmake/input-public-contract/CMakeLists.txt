cmake_minimum_required(VERSION 3.25)
project(RynUIInputPublicContract LANGUAGES CXX)
if(NOT DEFINED RYNUI_ROOT OR NOT DEFINED INPUT_CONTRACT_CASE)
    message(FATAL_ERROR "RYNUI_ROOT and INPUT_CONTRACT_CASE are required")
endif()
add_library(input_contract STATIC contract.cpp)
target_compile_features(input_contract PRIVATE cxx_std_20)
set_target_properties(input_contract PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(input_contract PRIVATE "${RYNUI_ROOT}/include")
target_compile_definitions(input_contract PRIVATE "INPUT_CASE=${INPUT_CONTRACT_CASE}")
